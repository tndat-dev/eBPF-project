% System overview of Sentinel Pulse (English version of thesis/figures/tikz-overview.tex). Styles in paper.tex.
% Top row: one worker node (data path left to right). Bottom row: master node (alert path right to left) and the
% offline fit. Absolute coordinates, anchor north west, so the boxes never overlap.
\begin{tikzpicture}[every node/.append style={anchor=north west}]
  % ---- one worker node
  \node[ext, text width=2.1cm, minimum height=2.3cm] (app) at (0,0) {\textbf{Application containers}\\namespace \texttt{production}; one cgroup each; unmodified};
  \node[kern, text width=3.7cm, minimum height=2.3cm] (probe) at (2.85,0) {\textbf{eBPF probe} (kernel)\\\texttt{sys\_enter}, \texttt{sys\_exit}, \texttt{\_\_seccomp\_filter}, \texttt{sched\_process\_fork}\\32-bit token per call; per-thread 4-grams counted per (cgroup, 0.5\,s window); lineage tag};
  \node[usr, text width=2.5cm, minimum height=2.3cm] (loader) at (7.3,0) {\textbf{Loader}\\every 0.5\,s drains the closed windows; SHA-256 and kernel proof checked at start};
  \node[mlb, text width=3.7cm, minimum height=2.3cm] (worker) at (10.55,0) {\textbf{Scoring worker}\\per-workload 4-gram count LM (Witten--Bell); $s(W)$ = mean of the top surprisals; alarm if $s(W) > \tau_{\text{workload}}$; one window fires};
  \node[sto, text width=1.6cm, minimum height=2.3cm] (dec) at (15.0,0) {\textbf{Decisions}\\local disk; segments on NFS};
  \node[usr, text width=2.1cm] (res) at (0,-2.65) {\textbf{Resolver}\\pod UID $\to$ cgroup $\to$ workload};
  \node[grp, fit=(app)(probe)(loader)(worker)(dec)(res)] (wgrp) {};
  \node[grplabel, anchor=south west] at (wgrp.north west) {Each worker node ($\times$3)};
  % ---- master node
  \node[ext, text width=1.9cm, minimum height=1.6cm] (tet) at (2.85,-4.75) {\textbf{Tetragon}\\exec, fork and connect events};
  \node[alr, text width=3.2cm, minimum height=1.6cm] (rca) at (5.4,-4.75) {\textbf{RCA}\\one case per delivered alert: process tree, ranked suspects};
  \node[alr, text width=2.0cm, minimum height=1.6cm] (am) at (9.25,-4.75) {\textbf{Alertmanager}\\of the cluster};
  \node[alr, text width=3.7cm, minimum height=1.6cm] (bridge) at (11.95,-4.75) {\textbf{Alert bridge}\\follows alarm rows over SSH; one episode per container (gaps $<$ 15\,s); push to Alertmanager};
  \node[grp, fit=(tet)(rca)(am)(bridge)] (mgrp) {};
  \node[grplabel, anchor=south west] at (mgrp.north west) {Master node};
  % ---- offline fit
  \node[mlb, text width=2.1cm, minimum height=1.6cm] (fit) at (0,-4.75) {\textbf{Offline fit}\\normal capture $\to$ counts $\to$ 5-fold cross-fit $\to$ Gumbel $\tau$};
  % ---- arrows
  \draw[arr] (app.east) -- node[elab, above] {syscalls} (probe.west);
  \draw[arr] (probe.east) -- node[elab, above] {map} (loader.west);
  \draw[arr] (loader.east) -- node[elab, above] {windows} (worker.west);
  \draw[arr] (worker.east) -- (dec.west);
  \draw[arrd] (res.east) -| node[elab, pos=0.3, above] {allow-list, workload identity} (loader.south);
  \draw[arr] (dec.south) |- node[elab, pos=0.3, right] {alarms} (bridge.east);
  \draw[arr] (bridge.west) -- (am.east);
  \draw[arr] (am.west) -- node[elab, above] {delivered} (rca.east);
  \draw[arr] (tet.east) -- (rca.west);
  \draw[arrd] (fit.north) -- ($(fit.north)+(0,0.5)$) -| node[elab, pos=0.25, above] {\texttt{MODELS.json} (SHA-256): models, thresholds} ($(worker.south)+(-1.2,0)$);
\end{tikzpicture}
