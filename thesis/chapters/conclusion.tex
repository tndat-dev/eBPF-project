\chapter{Kết luận và hướng phát triển}\label{chap:conclusion}

Một mô hình ngôn ngữ đơn giản trên token syscall, học chỉ từ normal, phát hiện được exploit thật của dataset công khai và tấn công trên cụm thật với độ trễ dưới 1 giây, với probe eBPF đủ rẻ để chạy liên tục. Mô hình chưa phân biệt được ác ý với một process vô hại mới, và gánh cảnh báo của hệ thống cuối còn gấp khoảng 3,6 lần ngân sách.

\section{Đóng góp}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  \textbf{Một probe eBPF dựng token syscall ngay trong kernel:} tên, mã lỗi, chế độ mở và thư mục đầu (đi ngược cây dentry), cỡ đọc ghi lúc trả về, cùng đếm 4-gram theo thread, với đúng một token cho mỗi syscall đã được kiểm bằng proof.
\item
  \textbf{Một mô hình học từ normal chạy giống hệt trên dataset và trên cụm:} mô hình ngôn ngữ Witten--Bell trên token, ngưỡng Gumbel trên điểm giữ ra của cross-fit, quyết định một cửa sổ. Trên LID-DS 2021 official nó đạt khoảng 97\% recall và 0,2--0,3\% báo nhầm; trong các nguồn đọc được, chưa thấy mô hình học nào đạt mức này trên đủ 15 kịch bản.
\item
  \textbf{Một hệ thống chạy live trên Kubernetes,} từ kernel tới Alertmanager và cây bằng chứng RCA, với mô hình, probe và code khoá bằng SHA-256.
\item
  \textbf{Một cách đánh giá live có đăng ký trước:} lượt 24--45 giờ trên tải thật, campaign có control idle và binary vô hại, đo độ trễ và gánh cảnh báo theo giờ, mọi con số trong đồ án tính lại được từ biên nhận.
\item
  \textbf{Bằng chứng thực nghiệm} rằng mô hình chuỗi học từ normal phát hiện điều mới chứ không phải ác ý: binary vô hại bị gắn cờ gần như luôn, và trên cả dataset lẫn cụm mô hình chưa hơn một luật novelty có ngưỡng.
\end{enumerate}

\section{Hướng phát triển}

Các hướng dưới được xếp theo giới hạn mà chúng nhắm tới (Chương 7).

\begin{itemize}
\tightlist
\item
  \textbf{Sự mới lạ không phải ác ý} (mục~\ref{sec:limits-novelty}):
  \begin{itemize}
  \tightlist
  \item
    chấm riêng các process đi vào container qua runtime exec bằng một mô hình riêng, học từ các lần exec vô hại, thay vì so chúng với normal của dịch vụ; điều kiện là không làm mất kết quả trên LID-DS 2021;
  \item
    giám sát tầng ứng dụng cho tấn công không đổi hành vi hệ thống (như \texttt{SQLInjectionUser} của LID-DS): uprobe eBPF vào hàm thực thi truy vấn của driver cơ sở dữ liệu, làm lớp bổ trợ cho Pulse.
  \end{itemize}
\item
  \textbf{Bằng chứng độc lập:} tìm nguyên nhân Heartbleed không chuyển sang LID-DS 2019 (cách trace ghi lỗi) mà không chỉnh trên chính dữ liệu đó; baseline học sâu (CHIDS, các mô hình của thư viện LID-DS) trên cùng bản ghi và cùng cách đo.
\item
  \textbf{Tính thực tế của tấn công:} exploit thật vào process của dịch vụ trên cụm, trong một namespace cô lập; một ứng dụng thứ hai có tải thay đổi suốt lượt chạy.
\item
  \textbf{Kẻ tấn công thích nghi:} thí nghiệm chia theo mức hiểu biết về detector, chỉ dùng 4-gram đã thấy, chạy chậm trong một process có sẵn; một trial \texttt{io\_\allowbreak{}uring}.
\item
  \textbf{Gánh cảnh báo:} học lại định kỳ chỉ từ normal đã được xác nhận; tách lệnh của exec probe (phần lớn đợt của lượt cuối) khỏi luồng session của người vận hành, ví dụ học riêng các lệnh probe khai báo trong pod spec; gom đợt thành sự cố chỉ giảm 223 đợt còn 172 và sự kiện Kubernetes chỉ giải thích 5 đợt, nên không đủ.
\item
  \textbf{Probe gọn hơn:} một bản build chỉ giữ chi tiết cho 5 syscall và bỏ bộ đếm theo cgroup, qua lượt thu và lượt live mới; đo chi phí trên request ở tải cao hơn 8 request/s.
\item
  \textbf{RCA:} đánh giá chất lượng case bởi người điều tra; LLM chạy cục bộ chỉ để chú thích case bất đồng bộ, ngoài đường cảnh báo, vì CORTEX mất trung vị 152,4 s mỗi cảnh báo \citep{wei2025}.
\end{itemize}
