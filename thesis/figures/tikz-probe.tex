  \node[kern, text width=3.6cm] (skip) at ($(enter.north east)+(0.35,0)$) {\textbf{\texttt{fexit/}}\\\textbf{\texttt{\_\_seccomp\_filter}}\\syscall bị seccomp bỏ qua không tới \texttt{sys\_exit}: phát token tên ở đây, bỏ đánh dấu chờ};
  \node[kern, text width=5.4cm] (exit) at ($(skip.north east)+(0.35,0)$) {\textbf{\texttt{raw\_tp/sys\_exit}} (39 syscall có hook)\\lỗi, nhóm 27 syscall: tên + errno\\đọc/ghi thành công: tên + $\min(15, \lfloor\log_2 \text{ret}\rfloor)$\\\texttt{open}, \texttt{openat}, \texttt{openat2}, \texttt{creat} thành công: tên + chế độ (ghi nếu \texttt{O\_WRONLY}, \texttt{O\_RDWR}, \texttt{O\_CREAT}, \texttt{O\_TRUNC} hoặc \texttt{O\_APPEND}) + thư mục đầu\\còn lại: tên};
  \node[kern, text width=5.4cm] (dir) at ($(exit.south west)+(0,-0.45)$) {\textbf{Thư mục đầu:} fd vừa mở $\to$ \texttt{struct file} $\to$ đi ngược \texttt{d\_parent}, vượt mount, tối đa 64 bước (\texttt{bpf\_loop}) tới gốc của task, tức gốc của container $\to$ băm FNV-1a thành phần đầu, giữ 20 bit; mã riêng cho \texttt{/} và đường dẫn tương đối};
